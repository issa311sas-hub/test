\begin{frame}
\frametitle{研究背景：ハルシネーションとキャリブレーション}
\begin{itemize}
	\item LLMは教育・業務・日常生活へ急速に普及
	\\⇒ \textcolor{blue}{ハルシネーション}（誤情報を正しい情報かのように生成する問題）は残っている[1]
	\vspace{10pt}
	\item 問題点：モデルが示す\textcolor{blue}{確信度}と実際の\textcolor{blue}{正答率}の乖離
	\\⇒ 確信度が正答率と整合していれば，利用者はリスク判断の手がかりにできる
	\vspace{10pt}
	\item この性質を\textcolor{red}{キャリブレーション}と呼び，ズレの大きさはECE等の指標で定量化される[2]
\end{itemize}
\end{frame}

\begin{frame}
\frametitle{研究背景：なぜプロンプト設計なのか}
\begin{itemize}
	\item キャリブレーションの改善手法
	\begin{itemize}
		\item Post-hoc補正
		\item ファインチューニング
		\item プロンプト設計
	\end{itemize}
	\vspace{10pt}
	\item GPT，Claude，Geminiなどの商用LLMは内部関数の計算値やパラメータへのアクセスを提供しない
	\\⇒ \textcolor{red}{モデルを問わずに実行可能な手段はプロンプト設計となる}
\end{itemize}
\end{frame}

\begin{frame}
\frametitle{研究背景：先行研究は英語タスクで有効性を示した}
\begin{itemize}
	\item Tian et al. 2023[3]／Xiong et al. 2024[4]
	\begin{itemize}
		\item 英語タスクにおいてプロンプト設計により確信度を言語で出力させる方式が
		      内部softmax確率より優れたキャリブレーション精度を達成
		\item \textcolor{red}{ECEを約50\%削減}
	\end{itemize}
\end{itemize}
\end{frame}

\begin{frame}
\frametitle{研究背景：MlingConfとの違い}
\begin{itemize}
	\item Xue et al. 2025[5]（MlingConf）は日本語含む5言語で同様の方式の有効性を確認
	\vspace{5pt}
	\item ただし3点の制限
	\begin{enumerate}
		\item 確信度の聞き出し方が1手法のみで方式間の比較がない
		\item 英語の問題を機械翻訳したデータセットのみを使用
		\item 検証モデルがGPT-3.5，Llama-3.1という旧世代に限られる
	\end{enumerate}
	\vspace{5pt}
	\\⇒ \textcolor{red}{日本語ネイティブなタスク上での方式間比較や現行世代モデルでの再現性検証は行われていない}
\end{itemize}
\end{frame}

\begin{frame}
\frametitle{研究目的}
\begin{block}{本研究の目的}
日本語タスクにおいて確信度プロンプト方式の違いがキャリブレーション性能の
評価指標にどのような差をもたらすか，またその効果が実験時点での現行世代の
商用LLMでも再現されるかを実証的に検証する
\end{block}
\vspace{10pt}
\begin{itemize}
	\item あわせて，モデル間・タスクドメイン間・日英言語間で改善パターンの
	      差についても分析する
\end{itemize}
\end{frame}

\begin{itemize}
	\item 複数の商用LLMに対して日本語タスク上で3種類の確信度プロンプト方式を
	      適用し，キャリブレーション精度を比較する実験を計画
	\vspace{10pt}
	\item \textbf{モデル}：GPT-4o，Claude Sonnet 4.6，Gemini 2.x，Swallow 等
	\vspace{5pt}
	\item \textbf{データセット}：MGSM（数学），JCommonsenseQA（常識），
	      JMMLU（知識），FLORES-200（翻訳）
\end{itemize}
\end{frame}

\begin{frame}
\frametitle{研究の方針：評価指標 ECE}
\begin{block}{$\mathrm{ECE}$(Expected Calibration Error)}
\[
\mathrm{ECE} = \sum_{m=1}^{M} \frac{|B_m|}{n} \left|\mathrm{acc}(B_m) - \mathrm{conf}(B_m)\right|
\]
\end{block}
\begin{itemize}
	\item $M$：ビン数，$B_m$：確信度を$m$番目のビンに分割した際の集合
	\item $n$：サンプル総数
	\item $\mathrm{acc}(B_m)$，$\mathrm{conf}(B_m)$：ビン$B_m$内の平均正答率・平均確信度
\end{itemize}
\end{frame}

\begin{frame}
\frametitle{研究の方針：評価指標 Brier Score}
\begin{block}{$\mathrm{Brier\ Score}$}
\[
\mathrm{Brier\ Score} = \frac{1}{N} \sum_{i=1}^{N} (p_i - y_i)^2
\]
\end{block}
\begin{itemize}
	\item $N$：問題数
	\item $p_i$：$i$番目の問題への予測確信度
	\item $y_i$：正誤（正解なら1，不正解なら0）
\end{itemize}
\end{frame}

\begin{frame}
\frametitle{研究の方針：評価指標 AUROC}
\begin{block}{$\mathrm{AUROC}$}
\[
\mathrm{AUROC} = \frac{1}{|\mathrm{Pos}||\mathrm{Neg}|} \sum_{p \in \mathrm{Pos}} \sum_{q \in \mathrm{Neg}}
\left( [p > q] + \frac{1}{2}[p = q] \right)
\]
\end{block}
\begin{itemize}
	\item $\mathrm{Pos}$：正解サンプルの確信度集合，$\mathrm{Neg}$：不正解サンプルの確信度集合
	\\⇒ \textcolor{red}{確信度が正誤をどれだけ識別できているかを測る}
\end{itemize}
\end{frame}

\begin{frame}
\frametitle{今後の予定：パイロット実験から本実験へ}
\begin{itemize}
	\item まず単一モデル・少数データでパイロット実験
	\begin{itemize}
		\item プロンプト出力やAPI呼び出しの安定性などスクリプトの動作確認
		\item 問題が見つかれば指示文・出力形式を修正し再確認
	\end{itemize}
	\vspace{10pt}
	\item 動作確認後，複数モデル×3方式×大規模な問題数で本実験
	\\⇒ \textcolor{red}{確信度プロンプト方式間の差と現行モデルでの再現性を検証}
\end{itemize}
\end{frame}

\begin{frame}
\frametitle{今後の予定：想定される結果パターン}
\begin{itemize}
	\item[(a)] 方式間に明確な差
	\\⇒ 統計検定で有意性を確認し，設計上の違いを考察
	\vspace{5pt}
	\item[(b)] 方式間に有意差なし
	\\⇒ 「聞き方の違いは性能に大きく影響しない」という知見を報告
	\vspace{5pt}
	\item[(c)] モデルにより効果が異なる
	\\⇒ 指示追従能力や学習データの違いとの関連を考察
\end{itemize}
\end{frame}

\begin{frame}
\frametitle{今後の予定：Murphy分解（調査中の追加実験）}
\begin{itemize}
	\item 実験データが十分に得られた場合，Brier ScoreのMurphy分解による
	      キャリブレーション精度と識別能力の分解を追加実験として検討
	\vspace{10pt}
	\item Murphy分解＝Brier Scoreを信頼度・分離度・不確実性に分解する統計手法
	\begin{itemize}
		\item 気象分野などで用いられる
	\end{itemize}
	\vspace{10pt}
	\item LLM分野でMurphy分解自体に焦点を当てた研究は少ない
	\\⇒ \textcolor{red}{同分野での利用可能性または利用されない背景を調査検討}
\end{itemize}
\end{frame}
